1% function to calculate new position and orientation by considering the
 2% actual coordiante frame
 3function [pd, rd_3x3] = Aktuelle_Position(thetaold, rd, pd, kord)
 4
 5pd_input = pd;
 6
 7ph=zeros(3,1);
 8l_endeffektor = [0.0; 0.0; 0.15]; %length endeffector
 9
10%%%%calculate with forward recursion formulars position till endeffector  %%%%%%%%%%%%
11a = [0 0.385 0.458 0.450];
12alpha = [ 1.5707963267948966 0.0 0.0 0.0];
13d = [0 0 0 0 ];
14
15thetaold_pos = thetaold(1:4);
16
17[x,y,z,pasterisk,jp] = ForwardRecursion(a,alpha,d,thetaold_pos);
18%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
19
20
21p=-jp; 
22ph(1:3,1)=p(1:3,5); %actual position
23
24 % calculate actual rotation with rotation matrices 
25 thetaold(2)=thetaold(2)-pi/2; 
26 thetaold(2:4)=-thetaold(2:4); 
27 RotZ1=[cos(thetaold(1)) -sin(thetaold(1)) 0;sin(thetaold(1)) cos(thetaold(1)) 0;0 0 1]; 
28 RotY2=[cos(thetaold(2)) 0 sin(thetaold(2));0 1 0;-sin(thetaold(2)) 0 cos(thetaold(2))]; 
29 RotY3=[cos(thetaold(3)) 0 sin(thetaold(3));0 1 0;-sin(thetaold(3)) 0 cos(thetaold(3))]; 
30 RotY4=[cos(thetaold(4)) 0 sin(thetaold(4));0 1 0;-sin(thetaold(4)) 0 cos(thetaold(4))]; 
31 RotY6=[cos(thetaold(6)) 0 sin(thetaold(6));0 1 0;-sin(thetaold(6)) 0 cos(thetaold(6))]; 
32 RotZ5=[cos(thetaold(5)) -sin(thetaold(5)) 0;sin(thetaold(5)) cos(thetaold(5)) 0;0 0 1]; 
33 RotZ7=[cos(thetaold(7)) -sin(thetaold(7)) 0;sin(thetaold(7)) cos(thetaold(7)) 0;0 0 1]; 
34 R07=RotZ1*RotY2*RotY3*RotY4*RotZ5*RotY6*RotZ7;
35 
36% calcualte endeffektor position
37 ph=ph+R07*l_endeffektor; 
38
39 
40 
41 
42 RZ=[cos(rd(1)) -sin(rd(1)) 0;sin(rd(1)) cos(rd(1)) 0;0 0 1];
43 RX=[1 0 0;0 cos(rd(2)) -sin(rd(2));0 sin(rd(2)) cos(rd(2))];
44 RY=[cos(rd(3)) 0 sin(rd(3));0 1 0;-sin(rd(3)) 0 cos(rd(3))];
45 
46 %calcualte desired endeffector rotation
47 R_des=R07*RZ*RY*RX; 
48
49 % consider coordinate frame and calcualte new position for CCD
50 if kord == 0
51 pd=ph+pd -R_des*l_endeffektor; %global coordiante frame
52 
53 max_pos = sqrt(pd(1)*pd(1)+pd(2)*pd(2)+pd(3)*pd(3))
54 
55 if (max_pos > 1.22)
56  pd=ph-pd_input -R_des*l_endeffektor 
57 end
58 
59 
60 elseif kord == 1
61 pd=ph+R_des*pd -R_des*l_endeffektor; %end-effector coordiante frame
62 
63 max_pos = sqrt(pd(1)*pd(1)+pd(2)*pd(2)+pd(3)*pd(3))
64 
65  if (max_pos > 1.22)
66  pd=ph-R_des*pd_input -R_des*l_endeffektor 
67 end
68 
69 end
70 
71 rd_3x3 = R_des; % new rotation matrix for CCD
72
73 
74end
