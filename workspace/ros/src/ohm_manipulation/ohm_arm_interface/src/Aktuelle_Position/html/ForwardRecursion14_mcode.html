1% Function Forward Recursion
 2function [x,y,z,pasterisk,jp] = ForwardRecursion(a,alpha,d,theta)
 3
 4% Forward Recursion formulas by L.T.Wang and B.Ravani to calculate the */
 5% forward kinematics of a manipulation arm. */
 6% a = link length */
 7% alpha = twist angle */
 8% d = link offset */
 9% theta = joint variable */
10n=length(a);
11x=[[1 0 0]',zeros(3,n)];
12y=[[0 1 0]',zeros(3,n)];
13z=[[0 0 1]',zeros(3,n)];
14pasterisk=zeros(3,n);
15jp=zeros(3,n+1);
16for i=1:n
17    
18    x(1:3,i+1)=x(1:3,i)*cos(theta(i))+y(1:3,i)*sin(theta(i));
19    z(1:3,i+1)=z(1:3,i)*cos(alpha(i))+cross(x(1:3,i+1),z(1:3,i))*sin(alpha(i));
20    y(1:3,i+1)=cross(z(1:3,i+1),x(1:3,i+1));
21    
22    pasterisk(1:3,i)=d(i)*z(1:3,i)+a(i)*x(1:3,i+1);
23    
24    jp(1:3,i+1)=jp(1:3,i)-pasterisk(1:3,i);
25end
26end
