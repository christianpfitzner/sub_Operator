1% Function for CCD Iteration
  2function [output] = CyclicCoordinateDescent(a,alpha,d,Rd,pd,theta,wo,alp,Epsilon)
  3d1=Rd(1:3,1);
  4d2=Rd(1:3,2);
  5d3=Rd(1:3,3);
  6thetaalt=zeros(1,7);
  7thetaalt(1:7)=theta(1:7);
  8theta=zeros(1,7);
  9theta(1:4)=thetaalt(1:4);
 10n=length(a);
 11iter=1;
 12
 13while 1
 14    
 15    [x,y,z,pasterisk,jp]=ForwardRecursion(a,alpha,d,theta); %call forward recursion function
 16    p=-jp;
 17    ph=p(1:3,n+1);
 18    phi=zeros(3,n);
 19    pdi=zeros(3,n);
 20    for c=1:n
 21        phi(1:3,c)=ph(1:3,1)-p(1:3,c);
 22        pdi(1:3,c)=pd(1:3)-p(1:3,c);
 23    end
 24    h1=x(1:3,n+1);
 25    h2=y(1:3,n+1);
 26    h3=z(1:3,n+1);
 27    deltap=norm(pd(1:3)-ph(1:3));
 28    Deltap=dot(deltap,deltap);
 29    % Deltao=wo(1)*(dot(d1,h1)-1)^2+wo(2)*(dot(d2,h2)-1)^2+wo(3)*(dot(d3,h3)-1)^2;
 30    % orientation error wont be considered, because of analytical solution of the orientation problem da
 31    
 32    
 33    Error=Deltap;  %+Deltao;
 34    if Error<Epsilon
 35        R_des=Rd;
 36        theta(2)=theta(2)-pi/2;
 37        theta(2:4)=-theta(2:4);
 38        % coordinate transformation to solve orientation problem
 39        
 40
 41    
 42
 43theta(5)=atan2(R_des(2,3)*cos(theta(1))-R_des(1,3)*sin(theta(1)),R_des(1,3)*cos(theta(2)+theta(3)+theta(4))*cos(theta(1))+R_des(2,3)*cos(theta(2)+theta(3)+theta(4))*sin(theta(1))-R_des(3,3)*sin(theta(2)+theta(3)+theta(4)));
 44      
 45
 46mp=R_des(1,3)*sin(theta(2)+theta(3)+theta(4))*cos(theta(1))+R_des(2,3)*sin(theta(2)+theta(3)+theta(4))*sin(theta(1))+R_des(3,3)*cos(theta(2)+theta(3)+theta(4))
 47     
 48
 49if mp^2>1 
 50theta(6)=thetaalt(6)  ;   
 51else 
 52theta(6)=atan2(sqrt(1-mp^2),mp);
 53end
 54
 55
 56
 57
 58theta(7)=atan2(R_des(1,2)*sin(theta(2)+theta(3)+theta(4))*cos(theta(1))+R_des(2,2)*sin(theta(2)+theta(3)+theta(4))*sin(theta(1))+R_des(3,2)*cos(theta(2)+theta(3)+theta(4)),-R_des(1,1)*sin(theta(2)+theta(3)+theta(4))*cos(theta(1))-R_des(2,1)*sin(theta(2)+theta(3)+theta(4))*sin(theta(1))-R_des(3,1)*cos(theta(2)+theta(3)+theta(4)));
 59  
 60if theta(7) < 0
 61    theta(7) = theta(7) +2*pi;
 62end
 63
 64%Winkel = [theta(5) theta(6) theta(7)]
 65
 66        theta(2:4)=-theta(2:4);
 67        theta(2)=theta(2)+pi/2;
 68        
 69        
 70        for i=5:7
 71            theta(i)=IK_CheckAngles(i,theta(i));
 72        end
 73        
 74        
 75        
 76        
 77        
 78        output=[theta,iter];
 79        %disp(['Cyclic Coordinate Descent successful after ',num2str(iter) ,' iterations!'])
 80        break
 81    else %  if line 35 - Error<Epsilon
 82        for i=n:-1:1
 83            wp=alp*(1+min([norm(pdi(1:3,i)),norm(phi(1:3,i))])/max([norm(pdi(1:3,i)),norm(phi(1:3,i))]));
 84            k1=wp*dot(pdi(1:3,i),z(1:3,i))*dot(phi(1:3,i),z(1:3,i))+wo(1)*dot(d1,z(1:3,i))*dot(h1,z(1:3,i))+wo(2)*dot(d2,z(1:3,i))*dot(h2,z(1:3,i))+wo(3)*dot(d3,z(1:3,i))*dot(h3,z(1:3,i));
 85            k2=wp*dot(pdi(1:3,i),phi(1:3,i))+wo(1)*dot(d1,h1)+wo(2)*dot(d2,h2)+wo(3)*dot(d3,h3);
 86            k3=dot(z(1:3,i),wp*cross(phi(1:3,i),pdi(1:3,i))+wo(1)*cross(h1,d1)+wo(2)*cross(h2,d2)+wo(3)*cross(h3,d3));
 87            phiasterisk=atan(-k3/(k1-k2));
 88            if (k1-k2)*cos(phiasterisk)-k3*sin(phiasterisk) >=0
 89                phiasterisk=phiasterisk+pi;
 90            end % if (ki-k2)...
 91            
 92            
 93            
 94            theta(i)=theta(i)+phiasterisk;
 95            
 96            
 97            % Check joint angles
 98            
 99            if theta(i)<=2*pi
100                if theta(i)>=-2*pi
101                    theta(i)=theta(i);
102                else
103                    theta(i)=theta(i)+2*pi;
104                end %if theta(i)>=-2*pi
105                
106            else
107                theta(i)=theta(i)-2*pi;
108            end %if theta(i)<=2*pi
109            
110            IK_CheckAngles(i,theta(i)); %check joint constraints / limits
111            
112            
113            
114            
115            
116            Rzphi=[z(1,i)^2*(1-cos(phiasterisk))+cos(phiasterisk) z(1,i)*z(2,i)*(1-cos(phiasterisk))-z(3,i)*sin(phiasterisk) z(1,i)*z(3,i)*(1-cos(phiasterisk))+z(2,i)*sin(phiasterisk);
117                z(1,i)*z(2,i)*(1-cos(phiasterisk))+z(3,i)*sin(phiasterisk) z(2,i)^2*(1-cos(phiasterisk))+cos(phiasterisk) z(2,i)*z(3,i)*(1-cos(phiasterisk))-z(1,i)*sin(phiasterisk);
118                z(1,i)*z(3,i)*(1-cos(phiasterisk))-z(2,i)*sin(phiasterisk) z(2,i)*z(3,i)*(1-cos(phiasterisk))+z(1,i)*sin(phiasterisk) z(3,i)^2*(1-cos(phiasterisk))+cos(phiasterisk)];
119            phi(1:3,i)=Rzphi*phi(1:3,i);
120            h1=Rzphi*h1;
121            h2=Rzphi*h2;
122            h3=Rzphi*h3;
123            if i>1
124                phi(1:3,i-1)=phi(1:3,i)+pasterisk(1:3,i-1);
125            end % if i>l
126        end % For-loop i=n:-1:1
127        iter=iter+1;
128        if iter>30
129            output=[thetaalt,iter];
130            %disp(['Termination after 200 iterations without achieving desired Epsilon. Error of ',num2str(Error),'!'])
131        end % if iter>30
132    end % if aus 35
133end %while
134end
